Practical recipes

Start from the task you have.

Small patterns you can copy, adapt, and dispose. Each recipe assumes const sculptor = new DomSculptor().

Create a card

const card = sculptor.tree({
  tag: "article",
  class: ["card"],
  attributes: { "aria-labelledby": "card-title" },
  children: [
    { tag: "h2", attributes: { id: "card-title" }, text: "Account" },
    { tag: "p", text: "Ready to use." }
  ]
});
sculptor.mount(card, "#app");

Enhance existing HTML

const panel = sculptor.wrap("[data-panel]");
const button = panel.child.find("button");
const open = sculptor.signal(false);

open.bindHidden(panel.child.find("[data-content]"), value => !value);
button.attr("aria-expanded", open);
button.on("click", () => open.update(value => !value));

Bind a form

const name = sculptor.signal("");
const input = sculptor.create("input")
  .attribute.set({ name: "name", autocomplete: "name" });

name.bind(input);
name.subscribe(value => saveDraft(value));

// Number parsing:

Render a keyed collection

const items = sculptor.signal([]);
const list = sculptor.create("ul");

items.list(list, {
  key: item => item.id,
  render: item => sculptor.create("li").setText(item.label),
  update: (row, item) => row.setText(item.label)
});

Render a large collection progressively

let list = sculptor.create("ul", "#app");

items.forEach(item => {
  sculptor.create("li", list).setText(item.label);
});

console.log(sculptor.rendering); // true while mounts remain queued

The first element mounts immediately and each later element mounts on a later animation frame. Creation stays chainable and does not require async code.

Load remote data

const users = sculptor.asyncState([]);
const status = sculptor.create("p").attribute.set("aria-live", "polite");

users.subscribe(snapshot => {
  if (snapshot.status === "loading") status.setText("Loading users…");
  if (snapshot.status === "success") status.setText(`${snapshot.data.length} users`);
  if (snapshot.status === "error") status.setText("Could not load users");
}, { immediate: true });

await users.run(({ signal }) =>
  fetch("/api/users", { signal }).then(response => response.json())
);

Own a route lifetime

let activeRoute = null;

function showRoute(factory) {
  activeRoute?.dispose();
  activeRoute = factory();
  sculptor.mount(activeRoute, "#app");
}

addEventListener("pagehide", () => activeRoute?.dispose(), { once: true });

Share a service with context

const Session = sculptor.createContextKey("Session");
const appContext = sculptor.createContext();
appContext.set(Session, sessionService);

const Account = sculptor.component((_props, context) => {
  const session = context.get(Session);
  return sculptor.create("p").setText(session.user.name);
});

const account = Account({}, appContext);

Clean up a feature

const scope = sculptor.createScope();

scope.run(() => {
  const query = sculptor.signal("");
  const input = sculptor.create("input");
  query.bind(input);
  sculptor.mount(input, "#app");

  const controller = new AbortController();
  scope.track(() => controller.abort());
});

scope.dispose(); // state, bindings, DOM, and custom abort
