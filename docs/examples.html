Practical, framework-free examples

Build it. Mount it. Clean it up.

Each focused example uses the immutable 3.0.0 browser ESM URL and includes its disposal path. Add the shared import below and the HTML or application values named by the example before running it.

import DomSculptor from "https://cdn.jsdelivr.net/npm/domsculptor@3.0.0/dist/domsculptor.esm.min.js";

Examples that mount to #app require <div id="app"></div>. Supply sourceRows for the table and save() for the form. Run the extension example where chrome.storage exists, and start progressive enhancement with matching server markup.

Counter

Signal-driven component

const sculptor = new DomSculptor();
const Counter = sculptor.component(({ initial = 0 }) => {
  const count = sculptor.signal(initial);
  return {
    root: sculptor.tree({
      tag: "button",
      text: count,
      on: { click: () => count.update(value => value + 1) }
    }),
    api: { count }
  };
});
const counter = Counter({ initial: 2 });
sculptor.mount(counter, "#app");
counter.dispose(); // page teardown

Keyed todo list

Stable rows and delegated removal

const scope = sculptor.createScope();
scope.run(() => {
  const todos = sculptor.signal([{ id: 1, text: "Ship safely" }]);
  const list = sculptor.create("ul");
  todos.list(list, {
    key: todo => todo.id,
    render: todo => sculptor.tree({
      tag: "li",
      attributes: { "data-id": todo.id },
      children: [{ tag: "span", text: todo.text }, { tag: "button", class: "delete", text: "Delete" }]
    }),
    update: (row, todo) => row.child.find("span").setText(todo.text)
  });
  list.on("click", ".delete", (_event, button) => {
    const id = Number(button.closest("li").dataset.id);
    todos.update(items => items.filter(todo => todo.id !== id));
  });
  sculptor.mount(list, "#app");
});
scope.dispose(); // rows, listeners, and subscriptions

Searchable table

Computed filtering and batching

const scope = sculptor.createScope();
scope.run(() => {
  const query = sculptor.signal("");
  const rows = sculptor.signal(sourceRows);
  const visible = sculptor.computed(
    () => rows.get().filter(row => row.name.toLowerCase().includes(query.get().toLowerCase())),
    [rows, query]
  );
  const input = sculptor.create("input");
  query.bind(input);
  const body = sculptor.create("tbody");
  visible.list(body, {
    key: row => row.id,
    render: row => sculptor.tree({ tag: "tr", children: [{ tag: "td", text: row.name }] })
  });
  sculptor.mount(sculptor.tree({ tag: "section", children: [input, { tag: "table", children: [body] }] }), "#app");
});
scope.dispose(); // computed, signals, bindings, and DOM

Accessible modal

Focus handoff and Escape cleanup

const previousFocus = document.activeElement;
const modal = sculptor.tree({
  tag: "section",
  attributes: { role: "dialog", "aria-modal": "true", "aria-labelledby": "modal-title" },
  children: [{ tag: "h2", attributes: { id: "modal-title" }, text: "Settings" }, { tag: "button", text: "Close" }]
});
const close = () => {
  modal.dispose();
  previousFocus?.focus();
};
modal.child.find("button").on("click", close);
modal.on("keydown", event => { if (event.key === "Escape") close(); });
sculptor.mount(modal, document.body);
modal.child.find("button").focus();
close(); // route teardown uses the same path

Validated form

Native validation and ARIA state

const scope = sculptor.createScope();
scope.run(() => {
  const email = sculptor.signal("");
  const invalid = sculptor.computed(() => !email.get().includes("@"), [email]);
  const input = sculptor.create("input").attribute.set({ type: "email", required: "" });
  email.bind(input);
  input.attr("aria-invalid", invalid).classToggle("invalid", invalid);
  const form = sculptor.tree({ tag: "form", children: [input, { tag: "button", text: "Save" }] });
  form.on("submit", event => {
    event.preventDefault();
    if (!invalid.get()) save(email.get());
  });
  sculptor.mount(form, "#app");
});
scope.dispose(); // validation, binding, listener, and form

Async user search

Cancellation and live status

const scope = sculptor.createScope();
scope.run(() => {
  const query = sculptor.signal("");
  const request = sculptor.asyncState([]);
  const input = sculptor.create("input");
  const status = sculptor.create("p").attribute.set({ "aria-live": "polite" });
  query.bind(input);
  input.on("input", async () => {
    const snapshot = await request.run(({ signal }) =>
      fetch(`/api/users?q=${encodeURIComponent(query.get())}`, { signal }).then(response => response.json())
    );
    if (snapshot.status === "success") status.setText(`${snapshot.data.length} users`);
    if (snapshot.status === "error") status.setText("Search failed");
  });
  sculptor.mount(sculptor.tree({ tag: "section", children: [input, status] }), "#app");
});
scope.dispose(); // aborts pending work and removes listeners

Browser-extension popup

Small, build-free controls

const scope = sculptor.createScope();
scope.run(() => {
  const enabled = sculptor.signal(false);
  const toggle = sculptor.create("input").attribute.set({ type: "checkbox" });
  enabled.bind(toggle);
  enabled.subscribe(value => chrome.storage.local.set({ enabled: value }), { immediate: true });
  sculptor.mount(sculptor.tree({
    tag: "label",
    children: [toggle, " Enable page tools"]
  }), "#app");
});
window.addEventListener("unload", () => scope.dispose(), { once: true });

Progressive enhancement

Adopt existing server markup

const scope = sculptor.createScope();
scope.run(() => {
  const menu = sculptor.adopt(document.querySelector("[data-menu]"));
  const open = sculptor.signal(false);
  const button = menu.child.find("button");
  const panel = menu.child.find("[data-panel]");
  open.bindHidden(panel, value => !value);
  button.attr("aria-expanded", open);
  button.on("click", () => open.update(value => !value));
  scope.track(() => sculptor.unmount(menu));
});
scope.dispose(); // detaches the server node and releases scoped listeners; the adopted wrapper remains reusable
